     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { useAuthStore, useOrdersStore } from '@/store';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Switch } from '@/components/ui/switch';
     8→import { Input } from '@/components/ui/input';
     9→import { Avatar, AvatarFallback } from '@/components/ui/avatar';
    10→import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
    11→import { 
    12→  Power, MapPin, Clock, Star, DollarSign, CheckCircle, 
    13→  Navigation, Phone, MessageCircle, Car, AlertCircle,
    14→  Wallet, TrendingUp, Calendar, LogOut, Settings, Home,
    15→  RefreshCw, Loader2, ArrowLeft, Crown, Edit, Bell, Banknote
    16→} from 'lucide-react';
    17→import { HideableBalanceDark, HideableBalanceLight } from '@/components/ui/hideable-balance';
    18→import type { Order, OrderStatus, Conversation, User } from '@/types';
    19→import { ChatView } from '@/components/chat/ChatView';
    20→import { toast } from 'sonner';
    21→import { parseJsonResponse } from '@/lib/json-helper';
    22→
    23→// Washer stats type
    24→interface WasherStats {
    25→  name: string;
    26→  rating: number;
    27→  totalRatings: number;
    28→  completedJobs: number;
    29→  totalEarnings: number;
    30→  todayEarnings: number;
    31→  todayJobs: number;
    32→  balance: number;
    33→}
    34→
    35→export function WasherApp() {
    36→  const { user, logout } = useAuthStore();
    37→  const { currentOrder, setCurrentOrder, orders, setOrders } = useOrdersStore();
    38→  const [activeTab, setActiveTab] = useState('dashboard');
    39→  const [isAvailable, setIsAvailable] = useState(true);
    40→  const [isLoading, setIsLoading] = useState(false);
    41→  const [pendingOrders, setPendingOrders] = useState<Order[]>([]);
    42→  const [showChat, setShowChat] = useState(false);
    43→  const [conversation, setConversation] = useState<Conversation | null>(null);
    44→  const [washerData, setWasherData] = useState<{
    45→    rating?: number;
    46→    totalRatings?: number;
    47→    completedJobs?: number;
    48→    totalEarnings?: number;
    49→    todayEarnings?: number;
    50→    todayJobs?: number;
    51→  } | null>(null);
    52→  const [isRefreshingBalance, setIsRefreshingBalance] = useState(false);
    53→  const [profileSection, setProfileSection] = useState<string | null>(null);
    54→
    55→  // Fetch conversation for current order
    56→  const fetchConversation = useCallback(async (orderId: string) => {
    57→    try {
    58→      const res = await fetch(`/api/conversations?orderId=${orderId}`);
    59→      const data = await parseJsonResponse<any>(res);
    60→      if (!data) return;
    61→      
    62→      if (data.success && data.conversation) {
    63→        setConversation(data.conversation);
    64→      }
    65→    } catch (error) {
    66→      console.error('Fetch conversation error:', error);
    67→    }
    68→  }, []);
    69→
    70→  // Open chat for order
    71→  const handleOpenChat = useCallback(async (order: Order) => {
    72→    if (order.id) {
    73→      await fetchConversation(order.id);
    74→      setShowChat(true);
    75→    }
    76→  }, [fetchConversation]);
    77→
    78→  // Fetch washer data (stats, balance)
    79→  const fetchWasherData = useCallback(async () => {
    80→    if (!user) return;
    81→    
    82→    setIsRefreshingBalance(true);
    83→    try {
    84→      const res = await fetch(`/api/washers/${user.id}`);
    85→      const data = await parseJsonResponse<any>(res);
    86→      if (!data) return;
    87→      
    88→      if (data.success && data.washer) {
    89→        setWasherData(data.washer);
    90→      }
    91→    } catch (error) {
    92→      console.error('Fetch washer data error:', error);
    93→    } finally {
    94→      setIsRefreshingBalance(false);
    95→    }
    96→  }, [user]);
    97→
    98→  // Get washer stats from washerData or defaults
    99→  const washerStats = {
   100→    name: user?.name || 'Laveur',
   101→    rating: washerData?.rating || 0,
   102→    totalRatings: washerData?.totalRatings || 0,
   103→    completedJobs: washerData?.completedJobs || 0,
   104→    totalEarnings: washerData?.totalEarnings || 0,
   105→    todayEarnings: washerData?.todayEarnings || 0,
   106→    todayJobs: washerData?.todayJobs || 0,
   107→    balance: washerData?.totalEarnings || 0,
   108→  };
   109→
   110→  // Fetch pending orders
   111→  const fetchPendingOrders = useCallback(async () => {
   112→    if (!user) return;
   113→    
   114→    setIsLoading(true);
   115→    try {
   116→      const res = await fetch(`/api/orders?userId=${user.id}&role=WASHER&status=PENDING`);
   117→      const data = await parseJsonResponse<any>(res);
   118→      if (!data) return;
   119→      
   120→      if (data.success) {
   121→        setPendingOrders(data.orders);
   122→      }
   123→    } catch (error) {
   124→      console.error('Fetch orders error:', error);
   125→    } finally {
   126→      setIsLoading(false);
   127→    }
   128→  }, [user]);
   129→
   130→  // Fetch my orders (active and history)
   131→  const fetchMyOrders = useCallback(async () => {
   132→    if (!user) return;
   133→    
   134→    try {
   135→      const res = await fetch(`/api/orders?userId=${user.id}&role=WASHER`);
   136→      const data = await parseJsonResponse<any>(res);
   137→      if (!data) return;
   138→      
   139→      if (data.success) {
   140→        setOrders(data.orders);
   141→        
   142→        // Set current active order
   143→        const active = data.orders.find((o: Order) => 
   144→          ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status)
   145→        );
   146→        if (active) {
   147→          setCurrentOrder(active);
   148→        }
   149→      }
   150→    } catch (error) {
   151→      console.error('Fetch my orders error:', error);
   152→    }
   153→  }, [user, setOrders, setCurrentOrder]);
   154→
   155→  // Initial load and polling
   156→  useEffect(() => {
   157→    fetchPendingOrders();
   158→    fetchMyOrders();
   159→    fetchWasherData();
   160→    
   161→    // Poll for new orders every 10 seconds when available
   162→    const interval = setInterval(() => {
   163→      if (isAvailable) {
   164→        fetchPendingOrders();
   165→      }
   166→    }, 10000);
   167→    
   168→    return () => clearInterval(interval);
   169→  }, [isAvailable, fetchPendingOrders, fetchMyOrders, fetchWasherData]);
   170→
   171→  // Accept order
   172→  const handleAcceptOrder = async (order: Order) => {
   173→    try {
   174→      const res = await fetch('/api/orders', {
   175→        method: 'PATCH',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify({
   178→          orderId: order.id,
   179→          status: 'ACCEPTED',
   180→          washerId: user?.id,
   181→        }),
   182→      });
   183→
   184→      const data = await parseJsonResponse<any>(res);
   185→      if (!data) return;
   186→
   187→      if (data.success) {
   188→        setCurrentOrder(data.order);
   189→        setPendingOrders(prev => prev.filter(o => o.id !== order.id));
   190→        setActiveTab('active');
   191→      }
   192→    } catch (error) {
   193→      console.error('Accept order error:', error);
   194→    }
   195→  };
   196→
   197→  // Update order status
   198→  const handleUpdateStatus = async (newStatus: OrderStatus) => {
   199→    if (!currentOrder) return;
   200→    
   201→    try {
   202→      // If this is a subscription order and completing, validate subscription first
   203→      if (currentOrder.isSubscriptionOrder && newStatus === 'COMPLETED') {
   204→        const res = await fetch('/api/subscriptions/validate', {
   205→          method: 'POST',
   206→          headers: { 'Content-Type': 'application/json' },
   207→          body: JSON.stringify({
   208→            orderId: currentOrder.id,
   209→            washerId: user?.id,
   210→            action: 'VALIDATE',
   211→          }),
   212→        });
   213→
   214→        const data = await parseJsonResponse<any>(res);
   215→        if (!data) return;
   216→
   217→        if (data.success) {
   218→          setCurrentOrder(null);
   219→          fetchMyOrders();
   220→          return;
   221→        }
   222→      }
   223→
   224→      // Regular status update
   225→      const res = await fetch('/api/orders', {
   226→        method: 'PATCH',
   227→        headers: { 'Content-Type': 'application/json' },
   228→        body: JSON.stringify({
   229→          orderId: currentOrder.id,
   230→          status: newStatus,
   231→        }),
   232→      });
   233→
   234→      const data = await parseJsonResponse<any>(res);
   235→      if (!data) return;
   236→
   237→      if (data.success) {
   238→        setCurrentOrder(data.order);
   239→        if (newStatus === 'COMPLETED') {
   240→          setCurrentOrder(null);
   241→          fetchMyOrders();
   242→        }
   243→      }
   244→    } catch (error) {
   245→      console.error('Update order error:', error);
   246→    }
   247→  };
   248→
   249→  return (
   250→    <div className="flex-1 flex flex-col bg-[#FAFAFA] overflow-hidden">
   251→      {/* Android Status Bar */}
   252→      <div className="h-6 bg-[#4CAF50] flex items-center justify-between px-4 flex-shrink-0 sticky top-0 z-50">
   253→        <span className="text-white text-xs font-medium">{new Date().toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })}</span>
   254→        <div className="flex items-center gap-1">
   255→          {/* Signal Network Bars */}
   256→          <div className="flex items-end gap-0.5">
   257→            <div className="w-1 h-1 bg-white rounded-sm"></div>
   258→            <div className="w-1 h-2 bg-white rounded-sm"></div>
   259→            <div className="w-1 h-3 bg-white rounded-sm"></div>
   260→            <div className="w-1 h-4 bg-white rounded-sm"></div>
   261→          </div>
   262→          {/* Battery */}
   263→          <div className="w-5 h-2.5 border border-white rounded-sm ml-1 relative">
   264→            <div className="absolute inset-0.5 bg-white rounded-sm" style={{ width: '70%' }}></div>
   265→          </div>
   266→        </div>
   267→      </div>
   268→
   269→      {/* Header with Availability */}
   270→      <div className="bg-white border-b px-4 py-3 flex-shrink-0 sticky top-6 z-40">
   271→        <div className="flex items-center justify-between">
   272→          <div className="flex items-center gap-3">
   273→            <div className="w-12 h-12 bg-gradient-to-br from-[#4CAF50] to-[#2E7D32] rounded-full flex items-center justify-center text-white text-xl font-bold">
   274→              {washerStats.name.charAt(0)}
   275→            </div>
   276→            <div>
   277→              <div className="font-semibold text-[#212121]">{washerStats.name}</div>
   278→              <div className="flex items-center gap-1">
   279→                {washerStats.rating > 0 ? (
   280→                  <>
   281→                    <Star className="w-4 h-4 text-[#FFC107] fill-[#FFC107]" />
   282→                    <span className="text-sm text-[#757575]">{washerStats.rating.toFixed(1)}</span>
   283→                  </>
   284→                ) : (
   285→                  <span className="text-sm text-[#9E9E9E]">Nouveau laveur</span>
   286→                )}
   287→              </div>
   288→            </div>
   289→          </div>
   290→          <div className="flex items-center gap-3">
   291→            <div className="text-right">
   292→              <div className="text-xs text-[#757575]">Disponibilité</div>
   293→              <div className={`text-sm font-medium ${isAvailable ? 'text-[#4CAF50]' : 'text-[#9E9E9E]'}`}>
   294→                {isAvailable ? 'En ligne' : 'Hors ligne'}
   295→              </div>
   296→            </div>
   297→            <Switch
   298→              checked={isAvailable}
   299→              onCheckedChange={setIsAvailable}
   300→              className={isAvailable ? 'bg-[#4CAF50]' : ''}
   301→            />
   302→          </div>
   303→        </div>
   304→      </div>
   305→
   306→      {/* Main Content */}
   307→      <div className="flex-1 overflow-y-auto pb-16">
   308→        {activeTab === 'dashboard' && (
   309→          <WasherDashboard 
   310→            stats={washerStats} 
   311→            isAvailable={isAvailable} 
   312→            isLoading={isLoading}
   313→            pendingOrders={pendingOrders}
   314→            onAccept={handleAcceptOrder}
   315→            onRefresh={fetchPendingOrders}
   316→            onRefreshBalance={fetchWasherData}
   317→            isRefreshingBalance={isRefreshingBalance}
   318→            acceptedOrders={orders.filter(o => ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status))}
   319→          />
   320→        )}
   321→        {activeTab === 'active' && (
   322→          <ActiveOrderView 
   323→            order={currentOrder} 
   324→            onUpdateStatus={handleUpdateStatus} 
   325→            onBack={() => setActiveTab('dashboard')}
   326→            onOpenChat={handleOpenChat}
   327→            acceptedOrders={orders.filter(o => ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status))}
   328→            onSelectOrder={setCurrentOrder}
   329→          />
   330→        )}
   331→        {activeTab === 'history' && (
   332→          <WasherOrderHistory 
   333→            orders={orders} 
   334→            onBack={() => setActiveTab('dashboard')} 
   335→          />
   336→        )}
   337→        {activeTab === 'earnings' && (
   338→          <WasherEarnings 
   339→            stats={washerStats} 
   340→            onBack={() => setActiveTab('dashboard')}
   341→            onRefreshBalance={fetchWasherData}
   342→            isRefreshingBalance={isRefreshingBalance}
   343→            washerId={user?.id || ''}
   344→          />
   345→        )}
   346→        {activeTab === 'profile' && (
   347→          profileSection ? (
   348→            <WasherProfileSection 
   349→              section={profileSection} 
   350→              onBack={() => setProfileSection(null)}
   351→              user={user}
   352→            />
   353→          ) : (
   354→            <WasherProfile 
   355→              user={user} 
   356→              stats={washerStats} 
   357→              onLogout={logout} 
   358→              onBack={() => setActiveTab('dashboard')}
   359→              onNavigate={setProfileSection}
   360→            />
   361→          )
   362→        )}
   363→      </div>
   364→
   365→      {/* Android Bottom Navigation - FIXED at bottom */}
   366→      <nav className="fixed bottom-0 left-0 right-0 bg-white border-t border-[#E0E0E0] flex justify-around items-center h-14 z-50 shadow-[0_-2px_10px_rgba(0,0,0,0.1)]">
   367→        {[
   368→          { id: 'dashboard', icon: Home, label: 'Accueil' },
   369→          { id: 'active', icon: Car, label: 'Active' },
   370→          { id: 'history', icon: Clock, label: 'Historique' },
   371→          { id: 'earnings', icon: Wallet, label: 'Revenus' },
   372→          { id: 'profile', icon: Settings, label: 'Profil' },
   373→        ].map((tab) => {
   374→          const isActive = activeTab === tab.id;
   375→          return (
   376→            <button
   377→              key={tab.id}
   378→              onClick={() => setActiveTab(tab.id)}
   379→              className={`flex flex-col items-center justify-center py-1.5 px-4 transition-all ${
   380→                isActive ? 'text-[#4CAF50]' : 'text-[#757575]'
   381→              }`}
   382→            >
   383→              <tab.icon className={`w-6 h-6 ${isActive ? 'fill-current' : ''}`} />
   384→              <span className="text-[10px] font-medium mt-0.5">{tab.label}</span>
   385→            </button>
   386→          );
   387→        })}
   388→      </nav>
   389→
   390→      {/* Chat Overlay */}
   391→      {showChat && conversation && (
   392→        <div className="absolute inset-0 z-[60] bg-[#FAFAFA]">
   393→          <ChatView 
   394→            conversation={conversation} 
   395→            onBack={() => setShowChat(false)} 
   396→          />
   397→        </div>
   398→      )}
   399→    </div>
   400→  );
   401→}

     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { useAuthStore, useOrdersStore } from '@/store';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Switch } from '@/components/ui/switch';
     8→import { Input } from '@/components/ui/input';
     9→import { Avatar, AvatarFallback } from '@/components/ui/avatar';
    10→import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
    11→import { 
    12→  Power, MapPin, Clock, Star, DollarSign, CheckCircle, 
    13→  Navigation, Phone, MessageCircle, Car, AlertCircle,
    14→  Wallet, TrendingUp, Calendar, LogOut, Settings, Home,
    15→  RefreshCw, Loader2, ArrowLeft, Crown, Edit, Bell, Banknote
    16→} from 'lucide-react';
    17→import { HideableBalanceDark, HideableBalanceLight } from '@/components/ui/hideable-balance';
    18→import type { Order, OrderStatus, Conversation, User } from '@/types';
    19→import { ChatView } from '@/components/chat/ChatView';
    20→import { toast } from 'sonner';
    21→
    22→// Washer stats type
    23→interface WasherStats {
    24→  name: string;
    25→  rating: number;
    26→  totalRatings: number;
    27→  completedJobs: number;
    28→  totalEarnings: number;
    29→  todayEarnings: number;
    30→  todayJobs: number;
    31→  balance: number;
    32→}
    33→
    34→export function WasherApp() {
    35→  const { user, logout } = useAuthStore();
    36→  const { currentOrder, setCurrentOrder, orders, setOrders } = useOrdersStore();
    37→  const [activeTab, setActiveTab] = useState('dashboard');
    38→  const [isAvailable, setIsAvailable] = useState(true);
    39→  const [isLoading, setIsLoading] = useState(false);
    40→  const [pendingOrders, setPendingOrders] = useState<Order[]>([]);
    41→  const [showChat, setShowChat] = useState(false);
    42→  const [conversation, setConversation] = useState<Conversation | null>(null);
    43→  const [washerData, setWasherData] = useState<{
    44→    rating?: number;
    45→    totalRatings?: number;
    46→    completedJobs?: number;
    47→    totalEarnings?: number;
    48→    todayEarnings?: number;
    49→    todayJobs?: number;
    50→  } | null>(null);
    51→  const [isRefreshingBalance, setIsRefreshingBalance] = useState(false);
    52→  const [profileSection, setProfileSection] = useState<string | null>(null);
    53→
    54→  // Fetch conversation for current order
    55→  const fetchConversation = useCallback(async (orderId: string) => {
    56→    try {
    57→      const res = await fetch(`/api/conversations?orderId=${orderId}`);
    58→      const data = await res.json();
    59→      
    60→      if (data.success && data.conversation) {
    61→        setConversation(data.conversation);
    62→      }
    63→    } catch (error) {
    64→      console.error('Fetch conversation error:', error);
    65→    }
    66→  }, []);
    67→
    68→  // Open chat for order
    69→  const handleOpenChat = useCallback(async (order: Order) => {
    70→    if (order.id) {
    71→      await fetchConversation(order.id);
    72→      setShowChat(true);
    73→    }
    74→  }, [fetchConversation]);
    75→
    76→  // Fetch washer data (stats, balance)
    77→  const fetchWasherData = useCallback(async () => {
    78→    if (!user) return;
    79→    
    80→    setIsRefreshingBalance(true);
    81→    try {
    82→      const res = await fetch(`/api/washers/${user.id}`);
    83→      const data = await res.json();
    84→      
    85→      if (data.success && data.washer) {
    86→        setWasherData(data.washer);
    87→      }
    88→    } catch (error) {
    89→      console.error('Fetch washer data error:', error);
    90→    } finally {
    91→      setIsRefreshingBalance(false);
    92→    }
    93→  }, [user]);
    94→
    95→  // Get washer stats from washerData or defaults
    96→  const washerStats = {
    97→    name: user?.name || 'Laveur',
    98→    rating: washerData?.rating || 0,
    99→    totalRatings: washerData?.totalRatings || 0,
   100→    completedJobs: washerData?.completedJobs || 0,
   101→    totalEarnings: washerData?.totalEarnings || 0,
   102→    todayEarnings: washerData?.todayEarnings || 0,
   103→    todayJobs: washerData?.todayJobs || 0,
   104→    balance: washerData?.totalEarnings || 0,
   105→  };
   106→
   107→  // Fetch pending orders
   108→  const fetchPendingOrders = useCallback(async () => {
   109→    if (!user) return;
   110→    
   111→    setIsLoading(true);
   112→    try {
   113→      const res = await fetch(`/api/orders?userId=${user.id}&role=WASHER&status=PENDING`);
   114→      const data = await res.json();
   115→      
   116→      if (data.success) {
   117→        setPendingOrders(data.orders);
   118→      }
   119→    } catch (error) {
   120→      console.error('Fetch orders error:', error);
   121→    } finally {
   122→      setIsLoading(false);
   123→    }
   124→  }, [user]);
   125→
   126→  // Fetch my orders (active and history)
   127→  const fetchMyOrders = useCallback(async () => {
   128→    if (!user) return;
   129→    
   130→    try {
   131→      const res = await fetch(`/api/orders?userId=${user.id}&role=WASHER`);
   132→      const data = await res.json();
   133→      
   134→      if (data.success) {
   135→        setOrders(data.orders);
   136→        
   137→        // Set current active order
   138→        const active = data.orders.find((o: Order) => 
   139→          ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status)
   140→        );
   141→        if (active) {
   142→          setCurrentOrder(active);
   143→        }
   144→      }
   145→    } catch (error) {
   146→      console.error('Fetch my orders error:', error);
   147→    }
   148→  }, [user, setOrders, setCurrentOrder]);
   149→
   150→  // Initial load and polling
   151→  useEffect(() => {
   152→    fetchPendingOrders();
   153→    fetchMyOrders();
   154→    fetchWasherData();
   155→    
   156→    // Poll for new orders every 10 seconds when available
   157→    const interval = setInterval(() => {
   158→      if (isAvailable) {
   159→        fetchPendingOrders();
   160→      }
   161→    }, 10000);
   162→    
   163→    return () => clearInterval(interval);
   164→  }, [isAvailable, fetchPendingOrders, fetchMyOrders, fetchWasherData]);
   165→
   166→  // Accept order
   167→  const handleAcceptOrder = async (order: Order) => {
   168→    try {
   169→      const res = await fetch('/api/orders', {
   170→        method: 'PATCH',
   171→        headers: { 'Content-Type': 'application/json' },
   172→        body: JSON.stringify({
   173→          orderId: order.id,
   174→          status: 'ACCEPTED',
   175→          washerId: user?.id,
   176→        }),
   177→      });
   178→
   179→      const data = await res.json();
   180→
   181→      if (data.success) {
   182→        setCurrentOrder(data.order);
   183→        setPendingOrders(prev => prev.filter(o => o.id !== order.id));
   184→        setActiveTab('active');
   185→      }
   186→    } catch (error) {
   187→      console.error('Accept order error:', error);
   188→    }
   189→  };
   190→
   191→  // Update order status
   192→  const handleUpdateStatus = async (newStatus: OrderStatus) => {
   193→    if (!currentOrder) return;
   194→    
   195→    try {
   196→      // If this is a subscription order and completing, validate subscription first
   197→      if (currentOrder.isSubscriptionOrder && newStatus === 'COMPLETED') {
   198→        const res = await fetch('/api/subscriptions/validate', {
   199→          method: 'POST',
   200→          headers: { 'Content-Type': 'application/json' },
   201→          body: JSON.stringify({
   202→            orderId: currentOrder.id,
   203→            washerId: user?.id,
   204→            action: 'VALIDATE',
   205→          }),
   206→        });
   207→
   208→        const data = await res.json();
   209→
   210→        if (data.success) {
   211→          setCurrentOrder(null);
   212→          fetchMyOrders();
   213→          return;
   214→        }
   215→      }
   216→
   217→      // Regular status update
   218→      const res = await fetch('/api/orders', {
   219→        method: 'PATCH',
   220→        headers: { 'Content-Type': 'application/json' },
   221→        body: JSON.stringify({
   222→          orderId: currentOrder.id,
   223→          status: newStatus,
   224→        }),
   225→      });
   226→
   227→      const data = await res.json();
   228→
   229→      if (data.success) {
   230→        setCurrentOrder(data.order);
   231→        if (newStatus === 'COMPLETED') {
   232→          setCurrentOrder(null);
   233→          fetchMyOrders();
   234→        }
   235→      }
   236→    } catch (error) {
   237→      console.error('Update order error:', error);
   238→    }
   239→  };
   240→
   241→  return (
   242→    <div className="flex-1 flex flex-col bg-[#FAFAFA] overflow-hidden">
   243→      {/* Android Status Bar */}
   244→      <div className="h-6 bg-[#4CAF50] flex items-center justify-between px-4 flex-shrink-0 sticky top-0 z-50">
   245→        <span className="text-white text-xs font-medium">{new Date().toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })}</span>
   246→        <div className="flex items-center gap-1">
   247→          {/* Signal Network Bars */}
   248→          <div className="flex items-end gap-0.5">
   249→            <div className="w-1 h-1 bg-white rounded-sm"></div>
   250→            <div className="w-1 h-2 bg-white rounded-sm"></div>
   251→            <div className="w-1 h-3 bg-white rounded-sm"></div>
   252→            <div className="w-1 h-4 bg-white rounded-sm"></div>
   253→          </div>
   254→          {/* Battery */}
   255→          <div className="w-5 h-2.5 border border-white rounded-sm ml-1 relative">
   256→            <div className="absolute inset-0.5 bg-white rounded-sm" style={{ width: '70%' }}></div>
   257→          </div>
   258→        </div>
   259→      </div>
   260→
   261→      {/* Header with Availability */}
   262→      <div className="bg-white border-b px-4 py-3 flex-shrink-0 sticky top-6 z-40">
   263→        <div className="flex items-center justify-between">
   264→          <div className="flex items-center gap-3">
   265→            <div className="w-12 h-12 bg-gradient-to-br from-[#4CAF50] to-[#2E7D32] rounded-full flex items-center justify-center text-white text-xl font-bold">
   266→              {washerStats.name.charAt(0)}
   267→            </div>
   268→            <div>
   269→              <div className="font-semibold text-[#212121]">{washerStats.name}</div>
   270→              <div className="flex items-center gap-1">
   271→                {washerStats.rating > 0 ? (
   272→                  <>
   273→                    <Star className="w-4 h-4 text-[#FFC107] fill-[#FFC107]" />
   274→                    <span className="text-sm text-[#757575]">{washerStats.rating.toFixed(1)}</span>
   275→                  </>
   276→                ) : (
   277→                  <span className="text-sm text-[#9E9E9E]">Nouveau laveur</span>
   278→                )}
   279→              </div>
   280→            </div>
   281→          </div>
   282→          <div className="flex items-center gap-3">
   283→            <div className="text-right">
   284→              <div className="text-xs text-[#757575]">Disponibilité</div>
   285→              <div className={`text-sm font-medium ${isAvailable ? 'text-[#4CAF50]' : 'text-[#9E9E9E]'}`}>
   286→                {isAvailable ? 'En ligne' : 'Hors ligne'}
   287→              </div>
   288→            </div>
   289→            <Switch
   290→              checked={isAvailable}
   291→              onCheckedChange={setIsAvailable}
   292→              className={isAvailable ? 'bg-[#4CAF50]' : ''}
   293→            />
   294→          </div>
   295→        </div>
   296→      </div>
   297→
   298→      {/* Main Content */}
   299→      <div className="flex-1 overflow-y-auto pb-16">
   300→        {activeTab === 'dashboard' && (
   301→          <WasherDashboard 
   302→            stats={washerStats} 
   303→            isAvailable={isAvailable} 
   304→            isLoading={isLoading}
   305→            pendingOrders={pendingOrders}
   306→            onAccept={handleAcceptOrder}
   307→            onRefresh={fetchPendingOrders}
   308→            onRefreshBalance={fetchWasherData}
   309→            isRefreshingBalance={isRefreshingBalance}
   310→            acceptedOrders={orders.filter(o => ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status))}
   311→          />
   312→        )}
   313→        {activeTab === 'active' && (
   314→          <ActiveOrderView 
   315→            order={currentOrder} 
   316→            onUpdateStatus={handleUpdateStatus} 
   317→            onBack={() => setActiveTab('dashboard')}
   318→            onOpenChat={handleOpenChat}
   319→            acceptedOrders={orders.filter(o => ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status))}
   320→            onSelectOrder={setCurrentOrder}
   321→          />
   322→        )}
   323→        {activeTab === 'history' && (
   324→          <WasherOrderHistory 
   325→            orders={orders} 
   326→            onBack={() => setActiveTab('dashboard')} 
   327→          />
   328→        )}
   329→        {activeTab === 'earnings' && (
   330→          <WasherEarnings 
   331→            stats={washerStats} 
   332→            onBack={() => setActiveTab('dashboard')}
   333→            onRefreshBalance={fetchWasherData}
   334→            isRefreshingBalance={isRefreshingBalance}
   335→            washerId={user?.id || ''}
   336→          />
   337→        )}
   338→        {activeTab === 'profile' && (
   339→          profileSection ? (
   340→            <WasherProfileSection 
   341→              section={profileSection} 
   342→              onBack={() => setProfileSection(null)}
   343→              user={user}
   344→            />
   345→          ) : (
   346→            <WasherProfile 
   347→              user={user} 
   348→              stats={washerStats} 
   349→              onLogout={logout} 
   350→              onBack={() => setActiveTab('dashboard')}
   351→              onNavigate={setProfileSection}
   352→            />
   353→          )
   354→        )}
   355→      </div>
   356→
   357→      {/* Android Bottom Navigation - FIXED at bottom */}
   358→      <nav className="fixed bottom-0 left-0 right-0 bg-white border-t border-[#E0E0E0] flex justify-around items-center h-14 z-50 shadow-[0_-2px_10px_rgba(0,0,0,0.1)]">
   359→        {[
   360→          { id: 'dashboard', icon: Home, label: 'Accueil' },
   361→          { id: 'active', icon: Car, label: 'Active' },
   362→          { id: 'history', icon: Clock, label: 'Historique' },
   363→          { id: 'earnings', icon: Wallet, label: 'Revenus' },
   364→          { id: 'profile', icon: Settings, label: 'Profil' },
   365→        ].map((tab) => {
   366→          const isActive = activeTab === tab.id;
   367→          return (
   368→            <button
   369→              key={tab.id}
   370→              onClick={() => setActiveTab(tab.id)}
   371→              className={`flex flex-col items-center justify-center py-1.5 px-4 transition-all ${
   372→                isActive ? 'text-[#4CAF50]' : 'text-[#757575]'
   373→              }`}
   374→            >
   375→              <tab.icon className={`w-6 h-6 ${isActive ? 'fill-current' : ''}`} />
   376→              <span className="text-[10px] font-medium mt-0.5">{tab.label}</span>
   377→            </button>
   378→          );
   379→        })}
   380→      </nav>
   381→
   382→      {/* Chat Overlay */}
   383→      {showChat && conversation && (
   384→        <div className="absolute inset-0 z-[60] bg-[#FAFAFA]">
   385→          <ChatView 
   386→            conversation={conversation} 
   387→            onBack={() => setShowChat(false)} 
   388→          />
   389→        </div>
   390→      )}
   391→    </div>
   392→  );
   393→}

   395→// Washer Dashboard

   399→  isLoading: boolean;

   487→      {/* Accepted Orders - Active tracking */}

   528→        <div className="flex items-center justify-between mb-3">
   529→          <h2 className="font-semibold text-[#212121]">Commandes en attente</h2>
   530→          <button 
   531→            onClick={onRefresh}
   532→            disabled={isLoading}
   533→            className="text-[#FF9800] text-sm flex items-center gap-1"
   534→          >
   535→            {isLoading ? (
   536→              <Loader2 className="w-4 h-4 animate-spin" />
   537→            ) : (
   538→              <RefreshCw className="w-4 h-4" />
   539→            )}
   540→          </button>
   541→        </div>
   542→
   543→        {isAvailable ? (
   544→          pendingOrders.length > 0 ? (
   545→            <div className="space-y-3">
   546→              {pendingOrders.map((order) => (
   547→                <Card key={order.id} className="border-0 shadow-sm overflow-hidden">
   548→                  <CardContent className="p-0">

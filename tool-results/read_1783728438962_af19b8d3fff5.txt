    34→    34→export function WasherApp() {
    35→    35→  const { user, logout } = useAuthStore();
    36→    36→  const { currentOrder, setCurrentOrder, orders, setOrders } = useOrdersStore();
    37→    37→  const [activeTab, setActiveTab] = useState('dashboard');
    38→    38→  const [isAvailable, setIsAvailable] = useState(true);
    39→    39→  const [isLoading, setIsLoading] = useState(false);

    41→    41→  const [showChat, setShowChat] = useState(false);

    44→    44→    rating?: number;
    45→    45→    totalRatings?: number;

    47→    47→    totalEarnings?: number;
    48→    48→    todayEarnings?: number;
    49→    49→    todayJobs?: number;

    51→    51→  const [isRefreshingBalance, setIsRefreshingBalance] = useState(false);

    67→    67→

    69→    69→  const handleOpenChat = useCallback(async (order: Order) => {

    73→    73→    }

    75→    75→
    76→    76→  // Fetch washer data (stats, balance)
    77→    77→  const fetchWasherData = useCallback(async () => {

    80→    80→    setIsRefreshingBalance(true);

    82→    82→      const res = await fetch(`/api/washers/${user.id}`);
    83→    83→      const data = await res.json();

    94→    94→
    95→    95→  // Get washer stats from washerData or defaults
    96→    96→  const washerStats = {
    97→    97→    name: user?.name || 'Laveur',
    98→    98→    rating: washerData?.rating || 0,
    99→    99→    totalRatings: washerData?.totalRatings || 0,
   100→   100→    completedJobs: washerData?.completedJobs || 0,
   101→   101→    totalEarnings: washerData?.totalEarnings || 0,
   102→   102→    todayEarnings: washerData?.todayEarnings || 0,
   103→   103→    todayJobs: washerData?.todayJobs || 0,
   104→   104→    balance: washerData?.totalEarnings || 0,
   105→   105→  };

   107→   107→  // Fetch pending orders

   109→   109→    if (!user) return;
   110→   110→    
   111→   111→    setIsLoading(true);
   112→   112→    try {
   113→   113→      const res = await fetch(`/api/orders?userId=${user.id}&role=WASHER&status=PENDING`);
   114→   114→      const data = await res.json();

   117→   117→        setPendingOrders(data.orders);
   118→   118→      }

   120→   120→      console.error('Fetch orders error:', error);
   121→   121→    } finally {
   122→   122→      setIsLoading(false);
   123→   123→    }
   124→   124→  }, [user]);
   125→   125→
   126→   126→  // Fetch my orders (active and history)
   127→   127→  const fetchMyOrders = useCallback(async () => {
   128→   128→    if (!user) return;
   129→   129→    
   130→   130→    try {
   131→   131→      const res = await fetch(`/api/orders?userId=${user.id}&role=WASHER`);
   132→   132→      const data = await res.json();
   133→   133→      
   134→   134→      if (data.success) {
   135→   135→        setOrders(data.orders);
   136→   136→        
   137→   137→        // Set current active order
   138→   138→        const active = data.orders.find((o: Order) => 
   139→   139→          ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status)
   140→   140→        );
   141→   141→        if (active) {
   142→   142→          setCurrentOrder(active);
   143→   143→        }
   144→   144→      }
   145→   145→    } catch (error) {
   146→   146→      console.error('Fetch my orders error:', error);
   147→   147→    }
   148→   148→  }, [user, setOrders, setCurrentOrder]);
   149→   149→
   150→   150→  // Initial load and polling
   151→   151→  useEffect(() => {
   152→   152→    fetchPendingOrders();
   153→   153→    fetchMyOrders();
   154→   154→    fetchWasherData();
   155→   155→    
   156→   156→    // Poll for new orders every 10 seconds when available
   157→   157→    const interval = setInterval(() => {
   158→   158→      if (isAvailable) {
   159→   159→        fetchPendingOrders();
   160→   160→      }
   161→   161→    }, 10000);
   162→   162→    
   163→   163→    return () => clearInterval(interval);
   164→   164→  }, [isAvailable, fetchPendingOrders, fetchMyOrders, fetchWasherData]);
   165→   165→
   166→   166→  // Accept order
   167→   167→  const handleAcceptOrder = async (order: Order) => {
   168→   168→    try {
   169→   169→      const res = await fetch('/api/orders', {
   170→   170→        method: 'PATCH',
   171→   171→        headers: { 'Content-Type': 'application/json' },
   172→   172→        body: JSON.stringify({
   173→   173→          orderId: order.id,
   174→   174→          status: 'ACCEPTED',
   175→   175→          washerId: user?.id,
   176→   176→        }),
   177→   177→      });
   178→   178→
   179→   179→      const data = await res.json();
   180→   180→
   181→   181→      if (data.success) {
   182→   182→        setCurrentOrder(data.order);
   183→   183→        setPendingOrders(prev => prev.filter(o => o.id !== order.id));
   184→   184→        setActiveTab('active');
   185→   185→      }

   187→   187→      console.error('Accept order error:', error);
   188→   188→    }

   196→   196→      // If this is a subscription order and completing, validate subscription first

   198→   198→        const res = await fetch('/api/subscriptions/validate', {
   199→   199→          method: 'POST',
   200→   200→          headers: { 'Content-Type': 'application/json' },
   201→   201→          body: JSON.stringify({
   202→   202→            orderId: currentOrder.id,
   203→   203→            washerId: user?.id,
   204→   204→            action: 'VALIDATE',
   205→   205→          }),
   206→   206→        });
   207→   207→
   208→   208→        const data = await res.json();
   209→   209→
   210→   210→        if (data.success) {
   211→   211→          setCurrentOrder(null);
   212→   212→          fetchMyOrders();
   213→   213→          return;
   214→   214→        }

   216→   216→
   217→   217→      // Regular status update
   218→   218→      const res = await fetch('/api/orders', {
   219→   219→        method: 'PATCH',
   220→   220→        headers: { 'Content-Type': 'application/json' },
   221→   221→        body: JSON.stringify({
   222→   222→          orderId: currentOrder.id,
   223→   223→          status: newStatus,
   224→   224→        }),
   225→   225→      });
   226→   226→
   227→   227→      const data = await res.json();
   228→   228→
   229→   229→      if (data.success) {
   230→   230→        setCurrentOrder(data.order);
   231→   231→        if (newStatus === 'COMPLETED') {
   232→   232→          setCurrentOrder(null);
   233→   233→          fetchMyOrders();
   234→   234→        }
   235→   235→      }
   236→   236→    } catch (error) {
   237→   237→      console.error('Update order error:', error);
   238→   238→    }
   239→   239→  };
   240→   240→
   241→   241→  return (
   242→   242→    <div className="flex-1 flex flex-col bg-[#FAFAFA] overflow-hidden">
   243→   243→      {/* Android Status Bar */}
   244→   244→      <div className="h-6 bg-[#4CAF50] flex items-center justify-between px-4 flex-shrink-0 sticky top-0 z-50">
   245→   245→        <span className="text-white text-xs font-medium">{new Date().toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })}</span>
   246→   246→        <div className="flex items-center gap-1">
   247→   247→          {/* Signal Network Bars */}
   248→   248→          <div className="flex items-end gap-0.5">
   249→   249→            <div className="w-1 h-1 bg-white rounded-sm"></div>
   250→   250→            <div className="w-1 h-2 bg-white rounded-sm"></div>
   251→   251→            <div className="w-1 h-3 bg-white rounded-sm"></div>
   252→   252→            <div className="w-1 h-4 bg-white rounded-sm"></div>
   253→   253→          </div>
   254→   254→          {/* Battery */}
   255→   255→          <div className="w-5 h-2.5 border border-white rounded-sm ml-1 relative">
   256→   256→            <div className="absolute inset-0.5 bg-white rounded-sm" style={{ width: '70%' }}></div>
   257→   257→          </div>
   258→   258→        </div>
   259→   259→      </div>
   260→   260→
   261→   261→      {/* Header with Availability */}
   262→   262→      <div className="bg-white border-b px-4 py-3 flex-shrink-0 sticky top-6 z-40">
   263→   263→        <div className="flex items-center justify-between">
   264→   264→          <div className="flex items-center gap-3">
   265→   265→            <div className="w-12 h-12 bg-gradient-to-br from-[#4CAF50] to-[#2E7D32] rounded-full flex items-center justify-center text-white text-xl font-bold">
   266→   266→              {washerStats.name.charAt(0)}
   267→   267→            </div>
   268→   268→            <div>
   269→   269→              <div className="font-semibold text-[#212121]">{washerStats.name}</div>
   270→   270→              <div className="flex items-center gap-1">
   271→   271→                {washerStats.rating > 0 ? (
   272→   272→                  <>
   273→   273→                    <Star className="w-4 h-4 text-[#FFC107] fill-[#FFC107]" />
   274→   274→                    <span className="text-sm text-[#757575]">{washerStats.rating.toFixed(1)}</span>
   275→   275→                  </>
   276→   276→                ) : (
   277→   277→                  <span className="text-sm text-[#9E9E9E]">Nouveau laveur</span>

   279→   279→              </div>
   280→   280→            </div>
   281→   281→          </div>
   282→   282→          <div className="flex items-center gap-3">

   295→   295→        </div>
   296→   296→      </div>
   297→   297→
   298→   298→      {/* Main Content */}

   309→   309→            isRefreshingBalance={isRefreshingBalance}
   310→   310→            acceptedOrders={orders.filter(o => ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status))}

   313→   313→        {activeTab === 'active' && (
   314→   314→          <ActiveOrderView 
   315→   315→            order={currentOrder}

   317→   317→            onBack={() => setActiveTab('dashboard')}
   318→   318→            onOpenChat={handleOpenChat}

   320→   320→            onSelectOrder={setCurrentOrder}
   321→   321→          />
   322→   322→        )}
   323→   323→        {activeTab === 'history' && (
   324→   324→          <WasherOrderHistory 
   325→   325→            orders={orders} 
   326→   326→            onBack={() => setActiveTab('dashboard')} 
   327→   327→          />
   328→   328→        )}
   329→   329→        {activeTab === 'earnings' && (
   330→   330→          <WasherEarnings 
   331→   331→            stats={washerStats} 
   332→   332→            onBack={() => setActiveTab('dashboard')}
   333→   333→            onRefreshBalance={fetchWasherData}
   334→   334→            isRefreshingBalance={isRefreshingBalance}
   335→   335→            washerId={user?.id || ''}
   336→   336→          />
   337→   337→        )}
   338→   338→        {activeTab === 'profile' && (
   339→   339→          profileSection ? (
   340→   340→            <WasherProfileSection 
   341→   341→              section={profileSection} 
   342→   342→              onBack={() => setProfileSection(null)}
   343→   343→              user={user}
   344→   344→            />
   345→   345→          ) : (
   346→   346→            <WasherProfile 
   347→   347→              user={user} 
   348→   348→              stats={washerStats} 
   349→   349→              onLogout={logout} 
   350→   350→              onBack={() => setActiveTab('dashboard')}
   351→   351→              onNavigate={setProfileSection}
   352→   352→            />
   353→   353→          )
   354→   354→        )}
   355→   355→      </div>
   356→   356→
   357→   357→      {/* Android Bottom Navigation - FIXED at bottom */}
   358→   358→      <nav className="fixed bottom-0 left-0 right-0 bg-white border-t border-[#E0E0E0] flex justify-around items-center h-14 z-50 shadow-[0_-2px_10px_rgba(0,0,0,0.1)]">
   359→   359→        {[
   360→   360→          { id: 'dashboard', icon: Home, label: 'Accueil' },
   361→   361→          { id: 'active', icon: Car, label: 'Active' },
   362→   362→          { id: 'history', icon: Clock, label: 'Historique' },

   371→   371→              className={`flex flex-col items-center justify-center py-1.5 px-4 transition-all ${

   380→   380→      </nav>

   383→   383→      {showChat && conversation && (
   384→   384→        <div className="absolute inset-0 z-[60] bg-[#FAFAFA]">
   385→   385→          <ChatView

   393→   393→}

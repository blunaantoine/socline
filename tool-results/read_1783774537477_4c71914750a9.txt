    35→    35→export function WasherApp() {
    36→    36→  const { user, logout } = useAuthStore();
    37→    37→  const { currentOrder, setCurrentOrder, orders, setOrders } = useOrdersStore();

    77→    77→
    78→    78→  // Fetch washer data (stats, balance)

   102→   102→    totalRatings: washerData?.totalRatings || 0,
   103→   103→    completedJobs: washerData?.completedJobs || 0,

   105→   105→    todayEarnings: washerData?.todayEarnings || 0,
   106→   106→    todayJobs: washerData?.todayJobs || 0,

   108→   108→  };
   109→   109→
   110→   110→  // Fetch pending orders

   112→   112→    if (!user) return;

   122→   122→      }

   124→   124→      console.error('Fetch orders error:', error);
   125→   125→    } finally {
   126→   126→      setIsLoading(false);
   127→   127→    }
   128→   128→  }, [user]);
   129→   129→
   130→   130→  // Fetch my orders (active and history)
   131→   131→  const fetchMyOrders = useCallback(async () => {
   132→   132→    if (!user) return;
   133→   133→    
   134→   134→    try {
   135→   135→      const res = await fetch(`/api/orders?userId=${user.id}&role=WASHER`);
   136→   136→      const data = await parseJsonResponse<any>(res);
   137→   137→      if (!data) return;
   138→   138→      
   139→   139→      if (data.success) {
   140→   140→        setOrders(data.orders);
   141→   141→        
   142→   142→        // Set current active order
   143→   143→        const active = data.orders.find((o: Order) => 
   144→   144→          ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status)

   147→   147→          setCurrentOrder(active);

   151→   151→      console.error('Fetch my orders error:', error);
   152→   152→    }

   154→   154→
   155→   155→  // Initial load and polling

   157→   157→    fetchPendingOrders();
   158→   158→    fetchMyOrders();
   159→   159→    fetchWasherData();
   160→   160→    
   161→   161→    // Poll for new orders every 10 seconds when available
   162→   162→    const interval = setInterval(() => {
   163→   163→      if (isAvailable) {
   164→   164→        fetchPendingOrders();

   167→   167→

   169→   169→  }, [isAvailable, fetchPendingOrders, fetchMyOrders, fetchWasherData]);
   170→   170→
   171→   171→  // Accept order
   172→   172→  const handleAcceptOrder = async (order: Order) => {
   173→   173→    try {
   174→   174→      const res = await fetch('/api/orders', {
   175→   175→        method: 'PATCH',
   176→   176→        headers: { 'Content-Type': 'application/json' },
   177→   177→        body: JSON.stringify({
   178→   178→          orderId: order.id,
   179→   179→          status: 'ACCEPTED',
   180→   180→          washerId: user?.id,
   181→   181→        }),

   183→   183→
   184→   184→      const data = await parseJsonResponse<any>(res);
   185→   185→      if (!data) return;

   205→   205→          method: 'POST',

   208→   208→            orderId: currentOrder.id,
   209→   209→            washerId: user?.id,
   210→   210→            action: 'VALIDATE',
   211→   211→          }),

   214→   214→        const data = await parseJsonResponse<any>(res);
   215→   215→        if (!data) return;

   218→   218→          setCurrentOrder(null);
   219→   219→          fetchMyOrders();

   226→   226→        method: 'PATCH',

   228→   228→        body: JSON.stringify({
   229→   229→          orderId: currentOrder.id,
   230→   230→          status: newStatus,
   231→   231→        }),
   232→   232→      });
   233→   233→
   234→   234→      const data = await parseJsonResponse<any>(res);
   235→   235→      if (!data) return;
   236→   236→
   237→   237→      if (data.success) {
   238→   238→        setCurrentOrder(data.order);
   239→   239→        if (newStatus === 'COMPLETED') {
   240→   240→          setCurrentOrder(null);
   241→   241→          fetchMyOrders();
   242→   242→        }
   243→   243→      }
   244→   244→    } catch (error) {
   245→   245→      console.error('Update order error:', error);
   246→   246→    }
   247→   247→  };
   248→   248→
   249→   249→  return (
   250→   250→    <div className="flex-1 flex flex-col bg-[#FAFAFA] overflow-hidden">
   251→   251→      {/* Android Status Bar */}

   253→   253→        <span className="text-white text-xs font-medium">{new Date().toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })}</span>
   254→   254→        <div className="flex items-center gap-1">
   255→   255→          {/* Signal Network Bars */}
   256→   256→          <div className="flex items-end gap-0.5">
   257→   257→            <div className="w-1 h-1 bg-white rounded-sm"></div>

   259→   259→            <div className="w-1 h-3 bg-white rounded-sm"></div>
   260→   260→            <div className="w-1 h-4 bg-white rounded-sm"></div>
   261→   261→          </div>
   262→   262→          {/* Battery */}

   264→   264→            <div className="absolute inset-0.5 bg-white rounded-sm" style={{ width: '70%' }}></div>
   265→   265→          </div>
   266→   266→        </div>
   267→   267→      </div>
   268→   268→
   269→   269→      {/* Header with Availability */}
   270→   270→      <div className="bg-white border-b px-4 py-3 flex-shrink-0 sticky top-6 z-40">
   271→   271→        <div className="flex items-center justify-between">
   272→   272→          <div className="flex items-center gap-3">
   273→   273→            <div className="w-12 h-12 bg-gradient-to-br from-[#4CAF50] to-[#2E7D32] rounded-full flex items-center justify-center text-white text-xl font-bold">
   274→   274→              {washerStats.name.charAt(0)}
   275→   275→            </div>
   276→   276→            <div>
   277→   277→              <div className="font-semibold text-[#212121]">{washerStats.name}</div>
   278→   278→              <div className="flex items-center gap-1">
   279→   279→                {washerStats.rating > 0 ? (
   280→   280→                  <>
   281→   281→                    <Star className="w-4 h-4 text-[#FFC107] fill-[#FFC107]" />
   282→   282→                    <span className="text-sm text-[#757575]">{washerStats.rating.toFixed(1)}</span>
   283→   283→                  </>

   314→   314→            onAccept={handleAcceptOrder}
   315→   315→            onRefresh={fetchPendingOrders}
   316→   316→            onRefreshBalance={fetchWasherData}
   317→   317→            isRefreshingBalance={isRefreshingBalance}
   318→   318→            acceptedOrders={orders.filter(o => ['ACCEPTED', 'EN_ROUTE', 'ARRIVED', 'IN_PROGRESS'].includes(o.status))}
   319→   319→          />
   320→   320→        )}
   321→   321→        {activeTab === 'active' && (
   322→   322→          <ActiveOrderView 
   323→   323→            order={currentOrder} 
   324→   324→            onUpdateStatus={handleUpdateStatus} 
   325→   325→            onBack={() => setActiveTab('dashboard')}
   326→   326→            onOpenChat={handleOpenChat}

   328→   328→            onSelectOrder={setCurrentOrder}
   329→   329→          />
   330→   330→        )}
   331→   331→        {activeTab === 'history' && (
   332→   332→          <WasherOrderHistory 
   333→   333→            orders={orders} 
   334→   334→            onBack={() => setActiveTab('dashboard')} 
   335→   335→          />
   336→   336→        )}
   337→   337→        {activeTab === 'earnings' && (
   338→   338→          <WasherEarnings 
   339→   339→            stats={washerStats} 
   340→   340→            onBack={() => setActiveTab('dashboard')}
   341→   341→            onRefreshBalance={fetchWasherData}
   342→   342→            isRefreshingBalance={isRefreshingBalance}
   343→   343→            washerId={user?.id || ''}
   344→   344→          />
   345→   345→        )}
   346→   346→        {activeTab === 'profile' && (
   347→   347→          profileSection ? (
   348→   348→            <WasherProfileSection 
   349→   349→              section={profileSection} 
   350→   350→              onBack={() => setProfileSection(null)}
   351→   351→              user={user}
   352→   352→            />
   353→   353→          ) : (
   354→   354→            <WasherProfile 
   355→   355→              user={user} 
   356→   356→              stats={washerStats} 
   357→   357→              onLogout={logout} 
   358→   358→              onBack={() => setActiveTab('dashboard')}
   359→   359→              onNavigate={setProfileSection}
   360→   360→            />
   361→   361→          )
   362→   362→        )}
   363→   363→      </div>
   364→   364→
   365→   365→      {/* Android Bottom Navigation - FIXED at bottom */}

   367→   367→        {[
   368→   368→          { id: 'dashboard', icon: Home, label: 'Accueil' },
   369→   369→          { id: 'active', icon: Car, label: 'Active' },
   370→   370→          { id: 'history', icon: Clock, label: 'Historique' },

   396→   396→          />
   397→   397→        </div>

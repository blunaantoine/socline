    67→    Array<{

    69→      name: string;
    70→      discountType: string;

    95→          setWalletBalance(data.wallet.balance);
    96→        }

    98→        console.error('Error fetching wallet:', error);
    99→      }
   100→    };
   101→    
   102→    fetchWallet();
   103→  }, [user?.id]);
   104→
   105→

   113→          const contentType = seedCheck.headers.get('content-type');
   114→          if (contentType && contentType.includes('application/json')) {
   115→            const seedData = await seedCheck.json();
   116→            
   117→            if (!seedData.seeded || seedData.servicesCount === 0) {
   118→              await fetch('/api/seed', { method: 'POST' });
   119→            }
   120→          }
   121→        }

   123→        // Fetch services
   124→        const res = await fetch('/api/services');

   127→        if (!resContentType || !resContentType.includes('application/json')) return;
   128→        const data = await res.json();

   131→          setServices(data.services);
   132→          
   133→          // Seed subscription plans after services are loaded
   134→          await fetch('/api/subscriptions/seed', { method: 'POST' }).catch(() => {});
   135→        }

   138→      }

   140→
   141→    loadServices();
   142→  }, [setServices]);
   143→

   162→

   170→        if (!contentType || !contentType.includes('application/json')) return;
   171→        const data = await res.json();
   172→        
   173→        if (data.success && data.promotions.length > 0) {
   174→          setPromotions(data.promotions);
   175→        }
   176→      } catch (error) {
   177→        console.error('Error fetching promotions:', error);
   178→      }
   179→    };
   180→    
   181→    fetchPromotions();
   182→  }, []);
   183→
   184→  // Auto-scroll promotions carousel

   186→    if (promotions.length <= 1) return;

   189→      setCurrentPromoIndex((prev) => (prev + 1) % promotions.length);
   190→    }, 4000); // Change every 4 seconds
   191→    
   192→    return () => clearInterval(interval);
   193→  }, [promotions]);
   194→
   195→  // Get user location
   196→  const getUserLocation = useCallback(() => {
   197→    setIsLoadingLocation(true);
   198→    if (navigator.geolocation) {
   199→      navigator.geolocation.getCurrentPosition(
   200→        (position) => {
   201→          setUserLocation({ latitude: position.coords.latitude, longitude: position.coords.longitude, address: 'Ma position' });
   202→          setIsLoadingLocation(false);
   203→        },
   204→        () => {
   205→          setUserLocation({ latitude: 6.1725, longitude: 1.2314, address: 'Lomé, Togo' });
   206→          setIsLoadingLocation(false);
   207→        }
   208→      );
   209→    } else {
   210→      setUserLocation({ latitude: 6.1725, longitude: 1.2314, address: 'Lomé, Togo' });
   211→      setIsLoadingLocation(false);
   212→    }
   213→  }, [setUserLocation]);
   214→
   215→  useEffect(() => {
   216→    const timer = setTimeout(getUserLocation, 0);
   217→    return () => clearTimeout(timer);
   218→  }, [getUserLocation]);
   219→
   220→  // Show auth screen if not authenticated
   221→  if (!isAuthenticated || !user) {
   222→    return (
   223→      <div className="flex-1 flex flex-col overflow-hidden relative bg-white">

   225→      </div>
   226→    );
   227→  }
   228→

   231→    return (

   234→        onBack={() => {

   242→  }
   243→
   244→  return (
   245→    <div className="flex-1 flex flex-col bg-[#FAFAFA] relative">

   247→      <div className="h-6 bg-[#FF9800] flex items-center justify-between px-4 flex-shrink-0">
   248→        <span className="text-white text-xs font-medium">{new Date().toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })}</span>
   249→        <div className="flex items-center gap-1">
   250→          {/* Signal Network Bars - de petite à grande */}

   263→
   264→      {/* Main Content */}
   265→      <div className="flex-1 flex flex-col overflow-hidden">
   266→        {/* App Bar (not shown on chat tab) */}
   267→        {activeTab !== 'chat' && activeTab !== 'booking' && (
   268→          <header className="bg-[#FF9800] px-4 py-3 flex items-center justify-between flex-shrink-0 shadow-md">

   270→              <button onClick={getUserLocation} className="flex items-center gap-2">

   272→                  <Loader2 className="w-5 h-5 text-white animate-spin" />
   273→                ) : (
   274→                  <MapPin className="w-5 h-5 text-white" />

   277→                  <p className="text-xs text-white/80">Position</p>

   282→            <div className="flex items-center gap-2">
   283→              <NotificationCenter />
   284→            </div>
   285→          </header>
   286→        )}
   287→

   291→            <HomeContent
   292→              services={services}

   294→              googleStations={googleStations}
   295→              isLoadingStations={isLoadingStations}

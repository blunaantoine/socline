   225→export function AuthScreen({ onComplete }: AuthScreenProps) {
   226→  const [mode, setMode] = useState<'welcome' | 'login' | 'register' | 'verify-otp' | 'washer-info'>('welcome');
   227→  const [isLoading, setIsLoading] = useState(false);
   228→  const [error, setError] = useState<string | null>(null);
   229→  const [showPin, setShowPin] = useState(false);
   230→  const [showConfirmPin, setShowConfirmPin] = useState(false);
   231→  
   232→  // Form fields
   233→  const [name, setName] = useState('');
   234→  const [phone, setPhone] = useState('');
   235→  const [plateNumber, setPlateNumber] = useState('');
   236→  const [carColor, setCarColor] = useState('');
   237→  const [pin, setPin] = useState('');
   238→  const [confirmPin, setConfirmPin] = useState('');
   239→  
   240→  // OTP fields
   241→  const [otp, setOtp] = useState('');
   242→  const [sentOtp, setSentOtp] = useState('');
   243→  const [resendTimer, setResendTimer] = useState(0);
   244→  
   245→  const { login } = useAuthStore();
   246→
   247→  // Timer for resend OTP
   248→  useEffect(() => {
   249→    if (resendTimer > 0) {
   250→      const timer = setTimeout(() => setResendTimer(resendTimer - 1), 1000);
   251→      return () => clearTimeout(timer);
   252→    }
   253→  }, [resendTimer]);
   254→
   255→  // Send OTP for registration
   256→  const handleSendOtp = async () => {
   257→    setError(null);
   258→    
   259→    if (!name.trim()) {
   260→      setError('Veuillez entrer votre nom');
   261→      return;
   262→    }
   263→    if (!phone.trim() || phone.length < 8) {
   264→      setError('Numéro de téléphone inval');
   265→      return;
   266→    }
   267→    if (!plateNumber.trim()) {
   268→      setError('Veuillez entrer le numéro de plaque');
   269→      return;
   270→    }
   271→    if (!carColor) {
   272→      setError('Veuillez sélectionner la couleur de votre voiture');
   273→      return;
   274→    }
   275→    if (pin.length !== 4) {
   276→      setError('Le PIN doit contenir 4 chiffres');
   277→      return;
   278→    }
   279→    if (confirmPin.length !== 4) {
   280→      setError('Veuillez confirmer votre PIN');
   281→      return;
   282→    }
   283→    if (pin !== confirmPin) {
   284→      setError('Les codes PIN ne correspondent pas');
   285→      return;
   286→    }
   287→
   288→    setIsLoading(true);
   289→    try {
   290→      const res = await fetch('/api/auth/send-otp', {
   291→        method: 'POST',
   292→        headers: { 'Content-Type': 'application/json' },
   293→        body: JSON.stringify({ phone }),
   294→      });
   295→      
   296→      if (!res.ok) {
   297→        setError('Erreur du serveur. Réessayez.');
   298→        return;
   299→      }
   300→      const contentType = res.headers.get('content-type');
   301→      if (!contentType || !contentType.includes('application/json')) {
   302→        setError('Réponse invalide du serveur');
   303→        return;
   304→      }
   305→      const data = await res.json();
   306→      
   307→      if (data.success) {
   308→        setSentOtp(data.otp || '123456');
   309→        setMode('verify-otp');
   310→        setResendTimer(60);
   311→      } else {
   312→        setError(data.error || 'Erreur lors de l\'envoi du code');
   313→      }
   314→    } catch {
   315→      setError('Erreur de connexion. Réessayez.');
   316→    } finally {
   317→      setIsLoading(false);
   318→    }
   319→  };
   320→
   321→  // Verify OTP and complete registration
   322→  const handleVerifyOtp = async () => {
   323→    setError(null);
   324→    
   325→    if (otp.length !== 6) {
   326→      setError('Le code doit contenir 6 chiffres');
   327→      return;
   328→    }
   329→
   330→    setIsLoading(true);
   331→    try {
   332→      // Send OTP along with registration data for verification
   333→      const res = await fetch('/api/auth/register', {
   334→        method: 'POST',
   335→        headers: { 'Content-Type': 'application/json' },
   336→        body: JSON.stringify({ name, phone, plateNumber, carColor, pin, otp }),
   337→      });
   338→      
   339→      if (!res.ok) {
   340→        setError('Erreur du serveur. Réessayez.');
   341→        return;
   342→      }
   343→      const contentType = res.headers.get('content-type');
   344→      if (!contentType || !contentType.includes('application/json')) {
   345→        setError('Réponse invalide du serveur');
   346→        return;
   347→      }
   348→      const data = await res.json();
   349→      
   350→      if (data.success) {
   351→        login(data.user, data.token);
   352→        onComplete();
   353→      } else {
   354→        setError(data.error || 'Erreur lors de l\'inscription');
   355→      }
   356→    } catch {
   357→      setError('Erreur de connexion. Réessayez.');
   358→    } finally {
   359→      setIsLoading(false);
   360→    }
   361→  };
   362→
   363→  const handleLogin = async () => {
   364→    setError(null);
   365→    
   366→    if (!phone.trim() || phone.length < 8) {
   367→      setError('Numéro de téléphone inval');
   368→      return;
   369→    }
   370→    if (pin.length !== 4) {
   371→      setError('PIN invalide');
   372→      return;
   373→    }
   374→
   375→    setIsLoading(true);
   376→    try {
   377→      const res = await fetch('/api/auth/login', {
   378→        method: 'POST',
   379→        headers: { 'Content-Type': 'application/json' },
   380→        body: JSON.stringify({ phone, pin }),
   381→      });
   382→      
   383→      if (!res.ok) {
   384→        setError('Erreur du serveur. Réessayez.');
   385→        return;
   386→      }
   387→      const contentType = res.headers.get('content-type');
   388→      if (!contentType || !contentType.includes('application/json')) {
   389→        setError('Réponse invalide du serveur');
   390→        return;
   391→      }
   392→      const data = await res.json();
   393→      
   394→      if (data.success) {
   395→        login(data.user, data.token);
   396→        onComplete();
   397→      } else {
   398→        setError(data.error || 'Erreur de connexion');
   399→      }
   400→    } catch {
   401→      setError('Erreur de connexion. Réessayez.');
   402→    } finally {
   403→      setIsLoading(false);
   404→    }
   405→  };
   406→
   407→  // Washer Info Screen
   408→  if (mode === 'washer-info') {
   409→    return <WasherRegistrationInfo onBack={() => setMode('welcome')} />;
   410→  }
   411→
   412→  // Welcome Screen
   413→  if (mode === 'welcome') {
   414→    return (
   415→      <div className="flex-1 flex flex-col bg-gradient-to-b from-[#FF9800] to-[#F57C00]">
   416→        {/* Logo Section */}
   417→        <div className="flex-1 flex flex-col items-center justify-center p-8">
   418→          <img 
   419→            src="/android-chrome-192x192.png"

   421→            className="w-24 h-24 mb-6 object-contain"
   422→          />
   423→          <h1 className="text-3xl font-bold text-white mb-2">Socline</h1>
   424→          <p className="text-white/80 text-center">Votre lavage auto à domicile</p>
   425→        </div>
   426→        
   427→        {/* Buttons */}
   428→        <div className="p-6 space-y-3">
   429→          <Button
   430→            onClick={() => setMode('login')}
   431→            className="w-full h-14 bg-white text-[#FF9800] hover:bg-white/90 rounded-2xl font-semibold text-lg"
   432→          >

   442→          {/* Discrete Washer Link */}

   446→          >
   447→            <span className="flex items-center justify-center gap-1">
   448→              Devenir partenaire laveur
   449→              <ChevronRight className="w-4 h-4" />
   450→            </span>
   451→          </button>
   452→        </div>
   453→      </div>
   454→    );
   455→  }
   456→
   457→  // Login Screen
   458→  if (mode === 'login') {
   459→    return (
   460→      <div className="flex-1 flex flex-col bg-[#FFF8F0]">
   461→        {/* Header */}
   462→        <div className="bg-[#FF9800] p-4 pt-8 pb-12 rounded-b-3xl">
   463→          <button onClick={() => setMode('welcome')} className="text-white mb-4">
   464→            <ArrowLeft className="w-6 h-6" />
   465→          </button>

   479→            
   480→            {/* Phone */}
   481→            <div className="space-y-2">
